# Rikugan: native IDA Pro / Binary Ninja plugin; the agent is libomc (../ohmyc), linked in-process.
# One host per build tree:
#   cmake -S . -B build/none  -G Ninja -DRIKUGAN_HOST=none  -DCMAKE_PREFIX_PATH=~/Qt/6.11.1/gcc_arm64   (tests)
#   cmake -S . -B build/binja -G Ninja -DRIKUGAN_HOST=binja -DCMAKE_PREFIX_PATH=~/Qt/6.11.1/gcc_arm64
#   cmake -S . -B build/ida   -G Ninja -DRIKUGAN_HOST=ida   -DCMAKE_PREFIX_PATH=~/Qt/6.8.2/gcc_arm64
# Windows: the same from an x64 Visual Studio developer prompt with the msvc2022_64 kits (tools\fetch_qt.ps1).
cmake_minimum_required(VERSION 3.25)
# before project(): MSVC's platform defaults would otherwise make an unset build type Debug (debug CRT, Qt6Cored.dll)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type of single-config generators")
endif()
project(rikugan VERSION 2.0.0 LANGUAGES C CXX)

set_property(CACHE RIKUGAN_HOST PROPERTY STRINGS binja ida none)
set(OMC_DIR "${CMAKE_SOURCE_DIR}/../ohmyc" CACHE PATH "omc (oh-my-c) source tree")
if(WIN32)
    set(BN_INSTALL_DIR "$ENV{LOCALAPPDATA}/Programs/Vector35/BinaryNinja" CACHE PATH "Binary Ninja install")
    set(IDA_DIR "$ENV{ProgramFiles}/IDA Professional 9.3" CACHE PATH "IDA Pro install")
else()
    set(BN_INSTALL_DIR "$ENV{HOME}/binaryninja" CACHE PATH "Binary Ninja install")
    set(IDA_DIR "$ENV{HOME}/ida-pro-9.3" CACHE PATH "IDA Pro install")
endif()
file(TO_CMAKE_PATH "${BN_INSTALL_DIR}" BN_INSTALL_DIR)
file(TO_CMAKE_PATH "${IDA_DIR}" IDA_DIR)

    set(_rk_tests_default ON)
else()
    set(_rk_tests_default OFF)

# Qt before 6.9 calls __yield() in qyieldcpu.h without including <arm_acle.h>. Clang on arm64 warns about the implicit
# declaration, and recent Apple clang refuses it.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64" AND Qt6_VERSION VERSION_LESS 6.9)
    add_compile_options("SHELL:-include arm_acle.h")
endif()

if(MSVC)
    # /utf-8: the sources and prompts are UTF-8; /Zc:__cplusplus and /permissive- are what Qt 6 requires
    add_compile_options(/utf-8 /permissive- /Zc:__cplusplus /EHsc /bigobj)
    # _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR: the host process may have loaded an older msvcp140.dll than this toolset's,
    # whose std::mutex would crash on a constexpr-constructed one
    add_compile_definitions(NOMINMAX _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
    set(RK_WARNINGS /W3)
else()
    set(RK_WARNINGS -Wall -Wextra -Wno-unused-parameter)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # GNU_UNIQUE objects (e.g. libstdc++'s make_shared tag in no-RTTI files) ignore the version script and would
    # stay exported from the plugin
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-fno-gnu-unique>)
endif()

function(rk_link_qt target)
    # binds the target to Qt for compiling (headers, moc); the host tree decides which .so files it links
    if(RIKUGAN_HOST STREQUAL "ida")
        foreach(_c Core Gui Widgets)
            target_include_directories(${target} PRIVATE $<TARGET_PROPERTY:Qt6::${_c},INTERFACE_INCLUDE_DIRECTORIES>)
            target_compile_definitions(${target} PRIVATE $<TARGET_PROPERTY:Qt6::${_c},INTERFACE_COMPILE_DEFINITIONS>)
        endforeach()
        # IDA's Qt is built with -qtnamespace QT (nm -D libQt6Core.so.6 shows N2QT7QString)
        target_compile_definitions(${target} PRIVATE QT_NAMESPACE=QT)
        set_target_properties(${target} PROPERTIES AUTOMOC_EXECUTABLE "$<TARGET_FILE:Qt6::moc>"
                                                   AUTORCC_EXECUTABLE "$<TARGET_FILE:Qt6::rcc>")
    else()
        target_link_libraries(${target} PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
    endif()
endfunction()
rk_link_qt(rikugan_common)

if(RIKUGAN_HOST STREQUAL "binja")
    set(BN_API_PATH "${CMAKE_SOURCE_DIR}/third_party/binaryninja-api")
    if(RIKUGAN_SDK_ONLY)
        # no install: the API links its generated core stubs, and the UI library's stand-in is made below
        set(HEADLESS ON)
        set(BN_ALLOW_STUBS ON CACHE BOOL "" FORCE)
        set(CMAKE_DISABLE_FIND_PACKAGE_BinaryNinjaCore ON)
    else()
        set(HEADLESS OFF)
    endif()
    add_subdirectory("${BN_API_PATH}" binaryninjaapi EXCLUDE_FROM_ALL SYSTEM)
    # what compiling against the UI headers needs (the API's binaryninjaui target, which also links the installed UI)
    add_library(rk_bn_ui INTERFACE)
    if(RIKUGAN_SDK_ONLY)
        target_include_directories(rk_bn_ui SYSTEM INTERFACE "${BN_API_PATH}/ui")
        target_link_libraries(rk_bn_ui INTERFACE binaryninjaapi Qt6::Core Qt6::Gui Qt6::Widgets)
    else()
        target_link_libraries(rk_bn_ui INTERFACE binaryninjaui Qt6::Core Qt6::Gui Qt6::Widgets)
    endif()
    file(GLOB_RECURSE RK_BN_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/binja/*.cpp" "${CMAKE_SOURCE_DIR}/src/binja/*.h")
    add_library(rikugan_binja_objects OBJECT ${RK_BN_SOURCES})
    target_include_directories(rikugan_binja_objects PRIVATE "${CMAKE_SOURCE_DIR}/src" "${OMC_INCLUDE_DIR}")
    # rikugan_common is linked as objects, so its PUBLIC usage requirements are applied by hand
    target_compile_definitions(rikugan_binja_objects PRIVATE $<TARGET_PROPERTY:rikugan_common,INTERFACE_COMPILE_DEFINITIONS>)
    target_compile_options(rikugan_binja_objects PRIVATE ${RK_WARNINGS})
    target_link_libraries(rikugan_binja_objects PRIVATE rk_bn_ui)
    set(_bn_objects $<TARGET_OBJECTS:rikugan_binja_objects> $<TARGET_OBJECTS:rikugan_common>)
    add_library(rikugan_binja MODULE ${_bn_objects})
    target_link_libraries(rikugan_binja PRIVATE rk_bn_ui)
    if(RIKUGAN_SDK_ONLY)
        rk_host_probe(rikugan_binja KIND binja OBJECTS ${_bn_objects} LIBS rk_bn_ui)
        if(MSVC)
            set(_bn_ui_lib "${RK_HOST_STUBS_DIR}/binaryninjaui.lib")
            add_custom_command(OUTPUT "${_bn_ui_lib}"
                COMMAND "${CMAKE_AR}" /nologo "/def:${RK_HOST_STUBS_DIR}/binaryninjaui.def" /machine:x64 "/out:${_bn_ui_lib}"
                DEPENDS "${RK_HOST_STUBS_DIR}/binaryninjaui.def"
                COMMENT "Import library of binaryninjaui.dll" VERBATIM)
            add_custom_target(rk_host_binaryninjaui_implib DEPENDS "${_bn_ui_lib}")
            add_dependencies(rikugan_binja rk_host_binaryninjaui_implib)
            target_link_libraries(rikugan_binja PRIVATE "${_bn_ui_lib}")
        else()
            if(APPLE)
                set(_bn_ui_name libbinaryninjaui.1.dylib)
            else()
                set(_bn_ui_name libbinaryninjaui.so.1)
            endif()
            rk_host_stub_library(rk_host_binaryninjaui SOURCE "${RK_HOST_STUBS_DIR}/binaryninjaui.c" NAME ${_bn_ui_name}
                                 INSTALL_NAME "@rpath/libbinaryninjaui.1.dylib" VERSIONS 1.0.0 0.0.0)
            target_link_libraries(rikugan_binja PRIVATE rk_host_binaryninjaui)
        endif()
    endif()
    rk_link_omc(rikugan_binja)
    # a MODULE is ".so" even on macOS; Binary Ninja and IDA load ".dylib" there. Windows: rikugan.dll
    if(WIN32)
        set(_bn_prefix "")
    else()
        set(_bn_prefix "lib")
    endif()
    set_target_properties(rikugan_binja PROPERTIES OUTPUT_NAME rikugan PREFIX "${_bn_prefix}"
                                                   SUFFIX "${CMAKE_SHARED_LIBRARY_SUFFIX}" SKIP_BUILD_RPATH ON)
    rk_export_only(rikugan_binja CorePluginABIVersion CorePluginInit UIPluginABIVersion UIPluginInit)
elseif(RIKUGAN_HOST STREQUAL "ida")
    set(IDA_SDK "${CMAKE_SOURCE_DIR}/third_party/ida-sdk/src")
    file(GLOB_RECURSE RK_IDA_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/ida/*.cpp" "${CMAKE_SOURCE_DIR}/src/ida/*.h")
    add_library(rikugan_ida_objects OBJECT ${RK_IDA_SOURCES})
    target_include_directories(rikugan_ida_objects PRIVATE "${CMAKE_SOURCE_DIR}/src" "${OMC_INCLUDE_DIR}")
    # the SDK's own warnings are not ours to fix (the Binary Ninja API is added as SYSTEM too)
    target_include_directories(rikugan_ida_objects SYSTEM PRIVATE "${IDA_SDK}/include")
    target_compile_definitions(rikugan_ida_objects PRIVATE $<TARGET_PROPERTY:rikugan_common,INTERFACE_COMPILE_DEFINITIONS>)
    if(MSVC)
        set(_ida_cxx_opts ${RK_WARNINGS})
    else()
        set(_ida_cxx_opts ${RK_WARNINGS} -fno-strict-aliasing)
    endif()
    target_compile_options(rikugan_ida_objects PRIVATE ${_ida_cxx_opts})
    if(WIN32)
        set(_ida_defs __EA64__=1 __NT__ __IDP__ NO_OBSOLETE_FUNCS)
    else()
        if(APPLE)
            set(_ida_os __MAC__)
        else()
            set(_ida_os __LINUX__)
        endif()
        set(_ida_defs __EA64__=1 ${_ida_os} __UNIX__ __IDP__ NO_OBSOLETE_FUNCS)
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
        list(APPEND _ida_defs __ARM__)
        if(APPLE)
            list(APPEND _ida_defs __APPLE_SILICON__)
        endif()
    endif()
    target_compile_definitions(rikugan_ida_objects PRIVATE ${_ida_defs})
    if(NOT MSVC)
        # files that subclass SDK types: the SDK libraries are built without RTTI (ELF/Mach-O typeinfo of their base
        # classes is never exported; MSVC emits RTTI in the module that uses it)
        file(GLOB_RECURSE RK_IDA_NORTTI CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/ida/sdk/*.cpp")
        set_source_files_properties(${RK_IDA_NORTTI} PROPERTIES COMPILE_OPTIONS "-fno-rtti")
    endif()
    rk_link_qt(rikugan_ida_objects)
    set(_ida_objects $<TARGET_OBJECTS:rikugan_ida_objects> $<TARGET_OBJECTS:rikugan_common>)
    add_library(rikugan_ida MODULE ${_ida_objects})
    if(WIN32)
        # the SDK ships import libraries for IDA itself and for its namespaced Qt
        set(_ida_lib "${IDA_SDK}/lib/x64_win_64/ida.lib")
        set(_idalib_lib "${IDA_SDK}/lib/x64_win_64/idalib.lib")
        set(_ida_qt "${IDA_SDK}/lib/x64_win_qt/Qt6Core.lib" "${IDA_SDK}/lib/x64_win_qt/Qt6Gui.lib"
                    "${IDA_SDK}/lib/x64_win_qt/Qt6Widgets.lib")
    elseif(RIKUGAN_SDK_ONLY)
        # no install: the SDK's libida, and stand-ins for IDA's Qt made from what the plugin uses
        if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
            set(_ida_arch arm64)
        else()
            set(_ida_arch x64)
        endif()
        if(APPLE)
            set(_ida_lib "${IDA_SDK}/lib/${_ida_arch}_mac_64/libida.dylib")
        else()
            set(_ida_lib "${IDA_SDK}/lib/${_ida_arch}_linux_64/libida.so")
        endif()
        rk_host_probe(rikugan_ida KIND ida OBJECTS ${_ida_objects} LIBS "${_ida_lib}"
                      QT_LIBS $<TARGET_FILE:Qt6::Core> $<TARGET_FILE:Qt6::Gui> $<TARGET_FILE:Qt6::Widgets>)
        set(_ida_qt)
        foreach(_m Core Gui Widgets)
            if(APPLE)
                set(_name Qt${_m})
            else()
                set(_name libQt6${_m}.so.6)
            endif()
            rk_host_stub_library(rk_host_qt6${_m} SOURCE "${RK_HOST_STUBS_DIR}/Qt6${_m}.c" NAME ${_name}
                                 INSTALL_NAME "@rpath/Qt${_m}.framework/Versions/A/Qt${_m}" VERSIONS 6.0.0 6.8.2)
            list(APPEND _ida_qt rk_host_qt6${_m})
        endforeach()
    elseif(APPLE)
        # IDA.app keeps its (namespaced) Qt as frameworks in Contents/Frameworks, next to Contents/MacOS
        set(_ida_qt)
        foreach(_m QtCore QtGui QtWidgets)
            find_library(IDA_${_m} NAMES ${_m} PATHS "${IDA_DIR}/../Frameworks" "${IDA_DIR}" NO_DEFAULT_PATH)
            if(NOT IDA_${_m})
                message(FATAL_ERROR "${_m}.framework not found in ${IDA_DIR}/../Frameworks (set -DIDA_DIR=<IDA.app>/Contents/MacOS)")
            endif()
            list(APPEND _ida_qt "${IDA_${_m}}")
        endforeach()
        set(_ida_lib "${IDA_DIR}/libida.dylib")
        set(_idalib_lib "${IDA_DIR}/libidalib.dylib")
    else()
        set(_ida_qt "${IDA_DIR}/libQt6Core.so.6" "${IDA_DIR}/libQt6Gui.so.6" "${IDA_DIR}/libQt6Widgets.so.6")
        set(_ida_lib "${IDA_DIR}/libida.so")
        set(_idalib_lib "${IDA_DIR}/libidalib.so")
    endif()
    target_link_libraries(rikugan_ida PRIVATE "${_ida_lib}" ${_ida_qt})
    rk_link_omc(rikugan_ida)
    set_target_properties(rikugan_ida PROPERTIES OUTPUT_NAME rikugan PREFIX "" SUFFIX "${CMAKE_SHARED_LIBRARY_SUFFIX}"
                                                 SKIP_BUILD_RPATH ON)
    if(APPLE)
        # IDA's @rpath libraries resolve against the app bundle, never against this build machine's paths
        set_target_properties(rikugan_ida PROPERTIES SKIP_BUILD_RPATH OFF BUILD_WITH_INSTALL_RPATH ON
                                                     INSTALL_RPATH "@executable_path;@executable_path/../Frameworks")
    endif()
    rk_export_only(rikugan_ida PLUGIN)
    # idalib executable that runs every IDA tool against a copy of /bin/ls (tests/ida/tool_cases.json). It runs on a
    # real IDA, so a build without the host leaves it out.
    if(NOT RIKUGAN_SDK_ONLY)
        file(GLOB RK_IDA_TOOLTEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tests/ida/*.cpp")
        set(RK_IDA_TOOL_SOURCES ${RK_IDA_SOURCES})
        list(REMOVE_ITEM RK_IDA_TOOL_SOURCES "${CMAKE_SOURCE_DIR}/src/ida/sdk/Plugin.cpp")
        add_executable(rikugan_ida_tooltest ${RK_IDA_TOOLTEST_SOURCES} ${RK_IDA_TOOL_SOURCES} $<TARGET_OBJECTS:rikugan_common>)
        target_include_directories(rikugan_ida_tooltest PRIVATE "${CMAKE_SOURCE_DIR}/src" "${OMC_INCLUDE_DIR}")
        target_include_directories(rikugan_ida_tooltest SYSTEM PRIVATE "${IDA_SDK}/include")
        target_compile_options(rikugan_ida_tooltest PRIVATE ${_ida_cxx_opts})
        target_compile_definitions(rikugan_ida_tooltest PRIVATE ${_ida_defs}
            $<TARGET_PROPERTY:rikugan_common,INTERFACE_COMPILE_DEFINITIONS>
            RK_TOOL_CASES="${CMAKE_SOURCE_DIR}/tests/ida/tool_cases.json")
        rk_link_qt(rikugan_ida_tooltest)
        target_link_libraries(rikugan_ida_tooltest PRIVATE "${_ida_lib}" "${_idalib_lib}" ${_ida_qt})
        rk_link_omc(rikugan_ida_tooltest)
        if(WIN32)
            # runs from the build tree with IDA's folder on PATH (ida.dll, idalib.dll, IDA's Qt); libomc.dll is next to it
            set_target_properties(rikugan_ida_tooltest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
        else()
            set_target_properties(rikugan_ida_tooltest PROPERTIES BUILD_RPATH "${IDA_DIR}")
        endif()
    endif()
elseif(NOT RIKUGAN_HOST STREQUAL "none")
    message(FATAL_ERROR "RIKUGAN_HOST must be binja, ida or none")
endif()
